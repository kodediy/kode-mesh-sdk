# The budget of internal RAM for this project, read by tools/check_ram.sh:
#
#   ../../tools/check_ram.sh build/mesh_target.elf ram_budget.txt
#
# This project links kode-sdk at the release the tree pins (v0.1.5) with every Mesh
# component on top (both layers), and CONFIG_KODEOS_RADIO=y as every app of the module needs
# it, so what it takes of the ESP32-P4's 640 KB of internal RAM at link time is what an app that
# uses the module starts with. Nearly all of it is kode-sdk's. The module's own statics come to
# about 13 KB: mesh_aos 11.7 KB (8 of them AO_Mesh's stack; the rest the owner's state, the
# received-packet ring, the two send slots, the events' queue of 16 and the listeners), mesh_drv
# 0.9 KB, mesh_sys 0.4 KB (the doors and the subscribers); mesh itself has none.
#
# static_max is the ceiling of that static image - code in IRAM, .data and .bss. Built from
# this tree against kode-sdk v0.1.5 it is some 229 KB, CONFIG_UART_ISR_IN_IRAM's driver among
# them, so the ceiling is some 26 KB above. tools/check_ram.sh prints the exact figure.
# A change that needs more either moves something to PSRAM or raises this number in the same
# commit, where a reviewer sees it. kode-sdk's own target keeps 364000 for itself.
static_max 256000

# Every data symbol of 2048 bytes or more in internal RAM, with the source
# file that defines it and the most it may be. A static this size that is not
# listed fails the check: a buffer belongs in PSRAM (khal_alloc_psram() in
# kode-sdk), and the ones below are here because something forces them to be
# internal. They are kode-sdk's and the libraries', listed at the size they
# have so that a pin that moves and grows one is seen.
#
# symbol               source file          max bytes

# AO_Mesh's stack: KQP_AO_STACK_BYTES, kode-sdk's default for an Active
# Object, internal for the reason kode-sdk's own are. The owner's deepest
# step is a radio command that waits on the chip; its high-water mark on a
# board may let it shrink.
l_stack                mesh_ao.c            8192

# The stacks of the system Active Objects (kqp_config.h's tiers). Internal
# because an AO may write flash, which takes PSRAM out of reach while the
# write lasts.
console_stack          console_ao.c         8192
storage_stack          storage_ao.c         6144
radio_stack            radio_ao.c           6144
power_stack            power_ao.c           4096
log_stack              log_ao.c             4096
ioexp_stack            ioexp_ao.c           4096
boot_stack             boot_ao.c            4096
autotest_stack         autotest_ao.c        4096
buttons_stack          buttons_ao.c         3072

# QP/C's three event pools.
s_jumbo_pool           kqp.c                16560
s_medium_pool          kqp.c                16128
s_large_pool           kqp.c                16128

# The console's Active Object. It grows with the verbs, so the line leaves
# room.
l_console              console_ao.c         4096

# What the libraries declare. Listed so that a version bump that grows one
# of them is seen.
xIsrStack              port.c               3072
s_watch_stack          khal_system_esp.c    3072
s_retention            sleep_retention.c    2724
